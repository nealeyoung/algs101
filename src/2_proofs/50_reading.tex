
\section{External sources on proofs for algorithms}\doHeader{External sources}

\begin{itemize}
\item
  \LLM
  Chapter 1 (proofs by contradiction, cases, proofs of ``if then'' statements, but nothing about ``for all'' statements),
  Chapter 6 (invariants),
  Chapter 5 (induction).

  Those topics are also covered in the videos of the first three lectures here:

  \parbox{0.9\textwidth}  {\url{https://ocw.mit.edu/courses/electrical-engineering-and-computer-science/6-042j-mathematics-for-computer-science-fall-2010/video-lectures/}}

\item \MF Chapters 2 (Logic), 3 (Proofs), and 11 (Algorithms)

\item More practice proof problems (marked ``P'') are at
  \url{https://www.cs.ucr.edu/~neal/puzzles}.

\item
  \emph{Book of Proof}, by Hammack, Parts II and III (about 70 pages with examples from discrete math).
  \url{http://www.people.vcu.edu/~rhammack/BookOfProof/}

\item
Long-form proofs are similar in spirit to Lamport's ``structured proofs,''
which you can read about here:
\url{https://www.microsoft.com/en-us/research/publication/write-21st-century-proof/}.

\end{itemize}
